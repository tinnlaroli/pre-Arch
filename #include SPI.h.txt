#include "SPI.h"
#include "Ethernet.h"
#include "sha1.h"
#include "mysql.h"
#include <Servo.h>

// Configuración de red
byte mac_addr[] = { 0x90, 0xA2, 0xDA, 0x0F, 0x69, 0xAC };
byte ip_addr[] = { 192, 168, 2, 127 };
byte netmask[] = { 255, 255, 255, 0 };
byte gateway_addr[] = { 192, 168, 2, 254 };
byte dns_addr[] = { 192, 168, 2, 254 };

IPAddress server_addr(192, 168, 2, 10);
Connector my_conn;

// Credenciales MySQL
char user[] = "prueba";
char password[] = "12345";

// Pines
const int pinPot = A1;      // Potenciómetro
const int pinPIR = A2;      // Sensor PIR
const int pinServoPot = A5; // Potenciómetro para el servo
const int pinServo = 9;     // Pin del servomotor
const int pinTrig = 7;      // Pin de disparo del ultrasónico
const int pinEcho = 6;      // Pin de eco del ultrasónico

Servo myServo;  // Objeto para el servo
int PULSOMIN = 500; 
int PULSOMAX = 2500;

// Buffers para consultas SQL
char query[128];
char valueStr[10];

// Variables para el sensor ultrasónico
int x; // Distancia en cm
int t; // Tiempo de viaje del sonido

// Variable para verificar la conexión a MySQL
bool conexionMySQL = false;

// Función para conectar a MySQL
bool conectarMySQL() {
  Serial.println("Conectando a MySQL...");
  if (my_conn.mysql_connect(server_addr, 3306, user, password)) {
      Serial.println("Conexión exitosa a MySQL!");
      return true;
  } else {
      Serial.println("Error al conectar a MySQL.");
      return false;
  }
}

// Función para leer el potenciómetro
int leerPotenciometro() {
    int valor = analogRead(pinPot);
    Serial.print("Valor del potenciómetro: ");
    Serial.println(valor);
    return valor;
}

// Función para leer el sensor PIR
int leerPIR() {
    int estado = analogRead(pinPIR); 
    Serial.print("Valor del sensor PIR: ");
    Serial.println(estado);
    return estado;
}

// Función para leer el potenciómetro del servo y moverlo
int controlarServo() {
    int valorPot = analogRead(pinServoPot);
    int angulo = map(valorPot, 0, 1023, 0, 180); 
    myServo.write(angulo);
    Serial.print("Ángulo del servo: ");
    Serial.println(angulo);
    return angulo;
}

// Función para leer el sensor ultrasónico y calcular la distancia
void leerUltrasonico() {
    digitalWrite(pinTrig, HIGH);
    delayMicroseconds(10);
    digitalWrite(pinTrig, LOW);

    t = pulseIn(pinEcho, HIGH); // Tiempo de viaje del sonido
    x = t / 58.2; // Convertir a distancia en cm

    Serial.print("Distancia ultrasónica: ");
    Serial.print(x);
    Serial.println(" cm");
}

// Función para insertar datos en la base de datos
void insertarDatos(char* nombreSensor, int numSensor, float valor) {
    dtostrf(valor, 1, 2, valueStr); // Convertir el valor a string con 2 decimales
    sprintf(query, "INSERT INTO arduino.sensor(mensaje, num_sensor, valor) VALUES ('%s', %d, %s);", nombreSensor, numSensor, valueStr);
    
    if (my_conn.cmd_query(query))
        Serial.print("Datos insertados correctamente: ");
        Serial.println(query);
}

void setup() {
    Ethernet.begin(mac_addr, ip_addr, dns_addr, gateway_addr, netmask);
    Serial.begin(9600);
    while (!Serial);

    myServo.attach(pinServo, PULSOMIN, PULSOMAX);  // Iniciar el servo
    myServo.write(0);
    pinMode(pinTrig, OUTPUT);
    pinMode(pinEcho, INPUT);

    delay(1000);
    Serial.println("Inicializando...");

    if (!conectarMySQL()) {
        Serial.println("Revisar conexión de red o credenciales.");
    }
}

void loop() {
    int valorPot = leerPotenciometro();
    insertarDatos("Potenciometro A1", 1, valorPot);

    int valorPIR = leerPIR();
    insertarDatos("Sensor PIR A2", 2, valorPIR);

    int anguloServo = controlarServo();
    insertarDatos("Servo Angle A5", 3, anguloServo);

    leerUltrasonico();
    insertarDatos("Ultrasonico D7- D6", 4, x);

  delay(5000);
}
