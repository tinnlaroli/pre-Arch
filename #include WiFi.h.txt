#include <WiFi.h>
#include <WiFiClient.h>
#include <ArduinoJson.h>
#include <Servo.h>

// Datos de la red WiFi
const char* ssid     = "INFINITUM91AC";
const char* password = "RMEnvy4Tyy";

// Datos del servidor de la API
const char* host = "192.168.1.104";
const int port = 3000;

// Definir pines para los LEDs y Servo
const int led1 = 5;
const int led2 = 6;
Servo miServo;
const int servoPin = 9;



// Función para probar el servo
void testServo() {
  Serial.println("Probando el servo...");
  
  // Mover de 0° a 180°
  for (int angulo = 0; angulo <= 180; angulo += 10) {
    miServo.write(angulo);
    delay(50);
  }
  
  // Mover de 180° a 0°
  for (int angulo = 180; angulo >= 0; angulo -= 10) {
    miServo.write(angulo);
    delay(50);
  }
  
  Serial.println("Prueba de servo completada");
}

// Función para probar los LEDs
void testLEDs() {
  Serial.println("Probando LEDs...");
  
  // Probar LED1
  Serial.println("LED1 ON");
  digitalWrite(led1, HIGH);
  delay(1000);
  Serial.println("LED1 OFF");
  digitalWrite(led1, LOW);
  delay(500);
  
  // Probar LED2
  Serial.println("LED2 ON");
  digitalWrite(led2, HIGH);
  delay(1000);
  Serial.println("LED2 OFF");
  digitalWrite(led2, LOW);
  delay(500);
  
  Serial.println("Prueba de LEDs completada");
}

// Función para conectar a WiFi
bool conectarWiFi() {
  Serial.print("Conectando a ");
  Serial.println(ssid);
  
  WiFi.begin(ssid, password);
  
  // Intentar conectar por 20 segundos
  unsigned long inicio = millis();
  
  while (WiFi.status() != WL_CONNECTED && millis() - inicio < 20000) {
    delay(500);
    Serial.print(".");
  }
  
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("\nError: No se pudo conectar a WiFi");
    return false;
  }
  
  Serial.println("\nConectado a WiFi");
  Serial.print("Dirección IP: ");
  Serial.println(WiFi.localIP());
  return true;
}

// Función para obtener la respuesta JSON del servidor
String obtenerJSON() {
  WiFiClient client;
  Serial.print("Conectando a ");
  Serial.print(host);
  Serial.print(":");
  Serial.println(port);
  
  if (!client.connect(host, port)) {
    Serial.println("Conexión fallida");
    return "";
  }
  
  // Realizar la petición GET
  String url = "/config";
  String request = String("GET ") + url + " HTTP/1.1\r\n" +
                   "Host: " + host + "\r\n" +
                   "Connection: close\r\n\r\n";
  client.print(request);
  Serial.println("Petición enviada");
  
  // Esperar respuesta del servidor
  unsigned long timeout = millis() + 5000;
  while (client.available() == 0) {
    if (millis() > timeout) {
      Serial.println("Tiempo de espera agotado");
      client.stop();
      return "";
    }
  }
  
  // Leer la respuesta del servidor
  String response;
  while (client.available()) {
    response += client.readString();
  }
  
  Serial.println("Respuesta recibida");
  client.stop();
  
  // Extraer solo la parte del JSON
  int jsonStart = response.indexOf("{");
  int jsonEnd = response.lastIndexOf("}");
  
  if (jsonStart == -1 || jsonEnd == -1) {
    Serial.println("Error: No se encontró JSON en la respuesta");
    return "";
  }
  
  String jsonString = response.substring(jsonStart, jsonEnd + 1);
  Serial.println("JSON extraído:");
  Serial.println(jsonString);
  
  return jsonString;
}

// Función para parsear JSON y controlar dispositivos
void controlarDispositivos(String jsonString) {
  if (jsonString.length() == 0) {
    Serial.println("No hay JSON para parsear");
    return;
  }
  
  // Parsear el JSON
  StaticJsonDocument<200> doc;
  DeserializationError error = deserializeJson(doc, jsonString);
  
  if (error) {
    Serial.print("Error parseando JSON: ");
    Serial.println(error.c_str());
    return;
  }
  
  // Leer los valores del JSON
  int estadoLed1 = doc["led1"];
  int estadoLed2 = doc["led2"];
  int anguloServo = doc["servo"];
  
  // Controlar los LEDs
  Serial.print("Activando LED1: ");
  Serial.println(estadoLed1);
  digitalWrite(led1, estadoLed1);
  
  Serial.print("Activando LED2: ");
  Serial.println(estadoLed2);
  digitalWrite(led2, estadoLed2);
  
  // Mover el servo
  Serial.print("Moviendo servo a: ");
  Serial.println(anguloServo);
  miServo.write(anguloServo);
}

// Función de inicialización
void setup() {
  Serial.begin(115200);
  delay(2000);
  Serial.println("\n--- Sistema iniciando ---");
  
  // Configurar pines como salida
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  
  // Inicializar el servo
  miServo.attach(servoPin);
  
  // Probar componentes primero
  Serial.println("Ejecutando pruebas de hardware...");
  testLEDs();
  testServo();
  
  // Conectar a la red WiFi
  if (!conectarWiFi()) {
    // Si falla la conexión, indicar error
    for (int i = 0; i < 5; i++) {
      digitalWrite(led1, HIGH);
      delay(100);
      digitalWrite(led1, LOW);
      delay(100);
    }
  }
}

// Función principal de ejecución
void loop() {
  // Verificar la conexión WiFi
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Conexión WiFi perdida. Reconectando...");
    conectarWiFi();
  }
  
  // Obtener JSON del servidor
  String jsonString = obtenerJSON();
  
  // Controlar dispositivos basados en el JSON
  controlarDispositivos(jsonString);
  
  // Espera antes de la próxima consulta
  delay(5000);
}
